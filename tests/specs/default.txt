== formats basic node ==
node   key="value"

[expect]
node key=value

== formats boolean with hash prefix ==
node   flag=#true

[expect]
node flag=#true

== formats children block ==
parent {
child   1
}

[expect]
parent {
    child 1
}

== keeps a leading block comment ==
/* regular block comment */
node   key="value"   flag=#true

[expect]
/* regular block comment */
node key=value flag=#true

== keeps a leading slashdash comment ==
/-regular_node 1 2
node   key="value"   flag=#true

[expect]
/-regular_node 1 2
node key=value flag=#true

== formats a document ==
package {
  name my-pkg
  version "1.2.3"

  dependencies {
    // Nodes can have standalone values as well as
    // key/value pairs.
    lodash "^3.2.1" optional=#true alias=underscore
  }

  scripts {
    // "Raw" and dedented multi-line strings are supported.
    message """
      hello
      world
      """
    build #"""
      echo "foo"
      node -c "console.log('hello, world!');"
      echo "foo" > some-file.txt
      """#
  }

  // `\` breaks up a single node across multiple lines.
  the-matrix 1 2 3 \
             4 5 6 \
             7 8 9

  // "Slashdash" comments operate at the node level,
  // with just `/-`.
  /-this-is-commented {
    this entire node {
      is gone
    }
  }
}

[expect]

        message "hello\nworld"

    }
    // `\` breaks up a single node across multiple lines.
    the-matrix 1 2 3 4 5 6 7 8 9
 // "Slashdash" comments operate at the node level,
  // with just `/-`.
  /-this-is-commented {
    this entire node {
      is gone
    }
  }
}
